### 날짜 형식 함수 (YYYY-MM-DD HH:mm:ss)
> 형식 글자를 바꿔 끼우는 작은 함수입니다. 라이브러리 없이 씁니다.
function format(date, pattern = 'YYYY-MM-DD HH:mm:ss') {
  const pad = n => String(n).padStart(2, '0');
  const map = {
    YYYY: date.getFullYear(), MM: pad(date.getMonth() + 1), DD: pad(date.getDate()),
    HH: pad(date.getHours()), mm: pad(date.getMinutes()), ss: pad(date.getSeconds()),
  };
  return pattern.replace(/YYYY|MM|DD|HH|mm|ss/g, k => map[k]);
}
console.log(format(new Date(2026, 10, 12, 9, 5, 3)));
console.log(format(new Date(2026, 10, 12), 'YYYY.MM.DD'));
console.log(format(new Date(2026, 10, 12), 'YYYYMMDD'));

### 글자 날짜를 Date 로 (YYYYMMDD · YYYY-MM-DD)
> DB 에서 온 '20261112' · '2026-11-12' 를 Date 로 바꿉니다. new Date('2026-11-12') 는 UTC 자정이라 쓰지 않습니다.
function parseDate(text) {
  const m = String(text).replace(/\D/g, '').match(/^(\d{4})(\d{2})(\d{2})(\d{2})?(\d{2})?(\d{2})?$/);
  if (!m) return null;
  const [, y, mo, d, h = 0, mi = 0, s = 0] = m;
  const date = new Date(+y, +mo - 1, +d, +h, +mi, +s);
  return date.getMonth() === +mo - 1 ? date : null; // 2월 30일 같은 없는 날은 null
}
console.log(parseDate('20261112'));
console.log(parseDate('2026-11-12 13:30:00'));
console.log(parseDate('2026-02-30'));

### 지난달 · 다음 달 같은 날
> 1월 31일에서 한 달 뒤는 2월 마지막 날로 맞춥니다(그냥 setMonth 하면 3월 3일이 됩니다).
function addMonths(date, n) {
  const d = new Date(date.getFullYear(), date.getMonth() + n, 1);
  const lastDay = new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
  d.setDate(Math.min(date.getDate(), lastDay));
  return d;
}

### 만 나이 계산
> 생일이 지나지 않았으면 한 살 뺍니다.
function age(birth, on = new Date()) {
  let a = on.getFullYear() - birth.getFullYear();
  const before = on.getMonth() < birth.getMonth() || (on.getMonth() === birth.getMonth() && on.getDate() < birth.getDate());
  if (before) a--;
  return a;
}

function addBusinessDays(date, n) {
  const d = new Date(date);
  while (n > 0) {
    d.setDate(d.getDate() + 1);
    if (d.getDay() !== 0 && d.getDay() !== 6 && !holidays.has(key(d))) n--;
  }
  return d;
}

### 두 날짜 사이 영업일 수
> 시작일 · 끝날을 모두 넣어 셉니다.
function businessDays(from, to) {
  let count = 0;
  for (const d = new Date(from); d <= to; d.setDate(d.getDate() + 1)) {
    if (d.getDay() !== 0 && d.getDay() !== 6) count++;
  }
  return count;
}
console.log(businessDays(new Date(2026, 10, 1), new Date(2026, 10, 30)));

function dateRange(from, to) {
  const out = [];

    out.push(d.toLocaleDateString('sv-SE')); // sv-SE 는 YYYY-MM-DD 모양

### 몇째 주인지 (ISO 주 번호)
> 주간 보고 번호. ISO 기준은 월요일 시작, 그 해 첫 목요일이 든 주가 1주입니다.
function isoWeek(date) {
  const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
  const day = d.getUTCDay() || 7;
  d.setUTCDate(d.getUTCDate() + 4 - day);
  const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
  return Math.ceil(((d - yearStart) / 86400000 + 1) / 7);
}
console.log(isoWeek(new Date(2026, 10, 12)));
console.log(isoWeek(new Date(2026, 0, 1)));

### 몇 분 전 · 몇 시간 전 (상대 시각)
> 게시판 · 알림의 「3분 전」. Intl.RelativeTimeFormat 을 씁니다.
function ago(date, now = new Date()) {
  const rtf = new Intl.RelativeTimeFormat('ko', { numeric: 'auto' });
  const sec = Math.round((date - now) / 1000);
  const units = [['year', 31536000], ['month', 2592000], ['day', 86400], ['hour', 3600], ['minute', 60]];
  for (const [unit, size] of units) {
    if (Math.abs(sec) >= size) return rtf.format(Math.trunc(sec / size), unit);
  }
  return '방금 전';
}
const now = new Date(2026, 10, 12, 12, 0);

### 걸린 시간을 「1시간 5분 3초」 로
> 초를 시 · 분 · 초로 나눕니다.
function duration(seconds) {
  const h = Math.floor(seconds / 3600);
  const m = Math.floor((seconds % 3600) / 60);
  const s = seconds % 60;
  return [h && `${h}시간`, m && `${m}분`, (s || !(h || m)) && `${s}초`].filter(Boolean).join(' ');
}
console.log(duration(3903), '/', duration(59), '/', duration(0));

### 기간 검색 조건 검사 (시작 ≤ 끝, 최대 3개월)
> 조회 단추를 누르기 전에 확인합니다.
function checkPeriod(from, to, maxMonths = 3) {
  if (!from || !to) return '기간을 넣어 주세요.';
  if (from > to) return '시작일이 종료일보다 늦습니다.';
  const limit = new Date(from);
  limit.setMonth(limit.getMonth() + maxMonths);
  if (to > limit) return `조회 기간은 ${maxMonths}개월까지입니다.`;
  return null;
}

console.log(checkPeriod(new Date(2026, 5, 1), new Date(2026, 0, 1)));

### 달력 한 달 칸 만들기 (6주 × 7일)
> 직접 그리는 달력. 앞뒤 달 날짜까지 42칸을 채웁니다.
function monthGrid(year, month) {
  const first = new Date(year, month, 1);
  const start = new Date(first);
  start.setDate(1 - first.getDay());
  const weeks = [];
  for (let w = 0; w < 6; w++) {
    const week = [];
    for (let i = 0; i < 7; i++) {
      const d = new Date(start);
      d.setDate(start.getDate() + w * 7 + i);
      week.push(d.getMonth() === month ? d.getDate() : '');
    }
    weeks.push(week);
  }
  return weeks;
}
console.table(monthGrid(2026, 10));

### 마감까지 남은 날 (D-day)
> 오늘이 마감 전이면 D-3, 지났으면 D+2.
function dday(deadline, today = new Date()) {
  const utc = d => Date.UTC(d.getFullYear(), d.getMonth(), d.getDate());
  const n = Math.round((utc(deadline) - utc(today)) / 86400000);
  return n === 0 ? 'D-day' : n > 0 ? `D-${n}` : `D+${-n}`;
}

function nthWeekday(year, month, weekday, n) {
  const first = new Date(year, month - 1, 1);
  const offset = (weekday - first.getDay() + 7) % 7;
  return new Date(year, month - 1, 1 + offset + (n - 1) * 7);
}

### 근속 연수 · 개월
> 입사일부터 오늘까지 「3년 4개월」.
function tenure(from, to) {
  let months = (to.getFullYear() - from.getFullYear()) * 12 + to.getMonth() - from.getMonth();
  if (to.getDate() < from.getDate()) months--;
  return `${Math.floor(months / 12)}년 ${months % 12}개월`;
}
console.log(tenure(new Date(2023, 6, 15), new Date(2026, 10, 12)));

function koreanMoney(n) {
  const units = [['조', 1e12], ['억', 1e8], ['만', 1e4]];
  let rest = Math.floor(n);
  const parts = [];
  for (const [name, size] of units) {
    if (rest >= size) {
      parts.push(Math.floor(rest / size).toLocaleString('ko-KR') + name);
      rest %= size;
    }
  }
  if (rest > 0 || !parts.length) parts.push(rest.toLocaleString('ko-KR'));
  return parts.join(' ') + ' 원';
}
console.log(koreanMoney(123456789), '/', koreanMoney(5000), '/', koreanMoney(3000000000000));

### 숫자를 한글로 (일금 오십만 원정)
> 견적서 · 영수증의 한글 금액.
function hangulNumber(n) {
  const digits = ['', '일', '이', '삼', '사', '오', '육', '칠', '팔', '구'];
  const small = ['', '십', '백', '천'];
  const big = ['', '만', '억', '조'];
  if (n === 0) return '영';
  let out = '';
  let group = 0;
  while (n > 0) {
    const part = n % 10000;
    if (part) {
      let text = '';
      String(part).padStart(4, '0').split('').forEach((c, i) => {
        if (c !== '0') text += digits[+c] + small[3 - i];
      });
      out = text + big[group] + out;
    }
    n = Math.floor(n / 10000);
    group++;
  }
  return out;
}
console.log(`일금 ${hangulNumber(500000)} 원정`);
console.log(hangulNumber(1234567890));

function split(total, count) {
  const base = Math.floor(total / count);
  const out = Array(count).fill(base);
  out[count - 1] += total - base * count;
  return out;
}

### 비율대로 나누기 (배분)
> 공통비를 부서 인원 비율로 나눕니다. 반올림 차이는 가장 큰 쪽에 몰아 합계를 맞춥니다.
function allocate(total, weights) {
  const sum = weights.reduce((s, w) => s + w, 0);
  const out = weights.map(w => Math.floor((total * w) / sum));
  const diff = total - out.reduce((s, n) => s + n, 0);
  out[weights.indexOf(Math.max(...weights))] += diff;
  return out;
}

function cutBytes(s, max) {
  let out = '';
  let size = 0;
  for (const ch of s) {
    const n = new TextEncoder().encode(ch).length;
    if (size + n > max) break;
    out += ch;
    size += n;
  }
  return out;
}

### 비슷한 글 찾기 (편집 거리)
> 오타 추천 · 비슷한 품명. 고칠 글자 수가 적을수록 비슷합니다.
function distance(a, b) {
  const dp = Array.from({ length: a.length + 1 }, (_, i) => [i, ...Array(b.length).fill(0)]);
  for (let j = 1; j <= b.length; j++) dp[0][j] = j;
  for (let i = 1; i <= a.length; i++) {
    for (let j = 1; j <= b.length; j++) {
      dp[i][j] = Math.min(dp[i - 1][j] + 1, dp[i][j - 1] + 1, dp[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1));
    }
  }
  return dp[a.length][b.length];
}
const words = ['brake', 'bracket', 'break', 'cable'];

### 은/는 · 이/가 · 을/를 조사 붙이기
> 받침 있는지 보고 조사를 고릅니다. 「사과를」, 「귤을」.
function josa(word, pair) {
  const [withFinal, without] = pair.split('/');
  const last = word.charCodeAt(word.length - 1);
  if (last < 0xac00 || last > 0xd7a3) return word + pair;
  const final = (last - 0xac00) % 28;
  const useFinal = pair === '으로/로' ? final !== 0 && final !== 8 : final !== 0;
  return word + (useFinal ? withFinal : without);
}

> 상태가 사용(Y)인 줄만 남깁니다.
console.table(data.filter(r => r.useYn === 'Y'));
--- 데이터
[{"code":"A","name":"사과","useYn":"Y"},{"code":"B","name":"배","useYn":"N"},{"code":"C","name":"감","useYn":"Y"}]

### 칸 하나만 뽑기 (map)
> 체크한 줄의 id 목록을 서버로 보낼 때.
console.log(data.map(r => r.id));
console.log(data.filter(r => r.checked).map(r => r.id));
--- 데이터
[{"id":11,"checked":true},{"id":12,"checked":false},{"id":13,"checked":true}]

### 첫 번째로 맞는 것 · 위치 (find · findIndex)
> 코드로 한 줄 찾기. 없으면 undefined · -1.
const row = data.find(r => r.code === 'B');
const index = data.findIndex(r => r.code === 'B');
console.log(row, index, data.find(r => r.code === 'Z'));
--- 데이터
[{"code":"A","name":"사과"},{"code":"B","name":"배"}]

### 하나라도 · 모두 (some · every)
> 체크한 줄이 있는지, 모두 저장됐는지.
console.log('체크 있음', data.some(r => r.checked));
console.log('모두 저장됨', data.every(r => r.saved));
--- 데이터
[{"checked":false,"saved":true},{"checked":true,"saved":false}]

### 여러 칸 기준 정렬
> 부서 오름차순, 같은 부서면 금액 내림차순.

### 원본을 바꾸지 않는 정렬 (toSorted)

### 줄 옮기기 (위로 · 아래로)

### 배열 섞기 (무작위)
> 시험 문제 순서 섞기. 피셔-예이츠 방법.
function shuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
console.log(shuffle([1, 2, 3, 4, 5, 6]));

### 그리드 머리글 눌러 정렬 (오름 · 내림 번갈아)
> 같은 칸을 다시 누르면 방향을 바꿉니다. 숫자 · 글 모두.
let sort = { key: null, asc: true };
function sortBy(rows, key) {
  sort = { key, asc: sort.key === key ? !sort.asc : true };
  const dir = sort.asc ? 1 : -1;
  return [...rows].sort((a, b) => {
    const x = a[key], y = b[key];
    if (x == null) return 1;
    if (y == null) return -1;
    return (typeof x === 'number' && typeof y === 'number' ? x - y : String(x).localeCompare(String(y), 'ko', { numeric: true })) * dir;
  });
}
console.table(sortBy(data, 'qty'));
console.table(sortBy(data, 'qty'));
--- 데이터
[{"item":"볼트","qty":10},{"item":"너트","qty":null},{"item":"와셔","qty":3},{"item":"핀","qty":25}]

### 깊게 합치기 (설정 객체)
> 안쪽 객체까지 합칩니다. 배열은 통째로 바꿉니다.
function deepMerge(target, source) {
  const out = { ...target };
  for (const [k, v] of Object.entries(source)) {
    out[k] = v && typeof v === 'object' && !Array.isArray(v) && typeof out[k] === 'object' ? deepMerge(out[k], v) : v;
  }
  return out;
}
console.log(deepMerge({ grid: { page: 1, size: 20 }, theme: 'light' }, { grid: { size: 50 }, cols: ['a'] }));

### 바뀐 칸만 찾기
> 저장 전 「바뀐 내용이 없습니다」 확인, 변경 이력 남기기.
function changed(before, after) {
  const out = {};
  for (const k of new Set([...Object.keys(before), ...Object.keys(after)])) {
    if (JSON.stringify(before[k]) !== JSON.stringify(after[k])) out[k] = { before: before[k], after: after[k] };
  }
  return out;
}
console.table(changed({ name: '가', qty: 1, memo: '' }, { name: '가', qty: 3, memo: '급함' }));

### 두 객체가 같은지 (깊은 비교)
> JSON.stringify 는 칸 순서가 다르면 다르다고 봅니다. 이 함수는 순서를 무시합니다.
function equal(a, b) {
  if (a === b) return true;
  if (typeof a !== 'object' || typeof b !== 'object' || !a || !b) return false;
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  const ka = Object.keys(a), kb = Object.keys(b);
  return ka.length === kb.length && ka.every(k => equal(a[k], b[k]));
}
console.log(equal({ a: 1, b: [1, { c: 2 }] }, { b: [1, { c: 2 }], a: 1 }), equal({ a: 1 }, { a: '1' }));

function set(o, path, value) {
  const keys = path.split('.');
  let cur = o;
  for (const k of keys.slice(0, -1)) cur = cur[k] ??= {};
  cur[keys.at(-1)] = value;
  return o;
}

### 오래된 것부터 버리는 캐시 (LRU)
> 크기를 정해 두고 넘치면 가장 오래 안 쓴 것을 버립니다. Map 의 넣은 순서를 씁니다.
class Lru {
  constructor(max) { this.max = max; this.map = new Map(); }
  get(k) {
    if (!this.map.has(k)) return undefined;
    const v = this.map.get(k);
    this.map.delete(k);
    this.map.set(k, v);
    return v;
  }
  set(k, v) {
    this.map.delete(k);
    this.map.set(k, v);
    if (this.map.size > this.max) this.map.delete(this.map.keys().next().value);
  }
}
const lru = new Lru(2);
lru.set('a', 1); lru.set('b', 2); lru.get('a'); lru.set('c', 3);
console.log([...lru.map.keys()]);

### 평평한 목록 → 트리 (부모 id)
> 메뉴 · 조직도. id 맵을 한 번 만들고 부모에 붙입니다. 부모가 없는 줄은 맨 위로.
function toTree(rows, idKey = 'id', parentKey = 'parentId') {
  const byId = new Map(rows.map(r => [r[idKey], { ...r, children: [] }]));
  const roots = [];
  for (const node of byId.values()) {
    const parent = byId.get(node[parentKey]);
    (parent ? parent.children : roots).push(node);
  }
  return roots;
}

function flatten(nodes, depth = 0, path = []) {
  const out = [];
  for (const n of nodes) {

    out.push({ depth, name: '  '.repeat(depth) + n.name, path: p.join(' > ') });

}

function findPath(nodes, id, path = []) {
  for (const n of nodes) {
    const p = [...path, n.name];
    if (n.id === id) return p;
    const found = findPath(n.children || [], id, p);
    if (found) return found;
  }
  return null;
}

### BOM 전개 (소요량 곱하기)
> 상위 품목 1개를 만드는 데 드는 하위 품목 수량. 아래로 내려갈수록 수량을 곱합니다.

function explode(item, qty = 1, level = 1, out = []) {
  for (const c of bom.get(item) || []) {
    const need = qty * c.qty;
    out.push({ level, item: '  '.repeat(level - 1) + c.child, qty: c.qty, need });
    explode(c.child, need, level + 1, out);
  }
  return out;
}

function pageButtons(current, last, around = 2) {
  const out = [];
  for (let p = 1; p <= last; p++) {
    if (p === 1 || p === last || Math.abs(p - current) <= around) out.push(p);
    else if (out.at(-1) !== '…') out.push('…');

}

### 사업자등록번호 검증 (검증 숫자)

function isBizNo(s) {
  const d = String(s).replace(/\D/g, '');
  if (d.length !== 10) return false;
  const w = [1, 3, 7, 1, 3, 7, 1, 3, 5];
  let sum = w.reduce((acc, x, i) => acc + x * +d[i], 0);
  sum += Math.floor((5 * +d[8]) / 10);
  return (10 - (sum % 10)) % 10 === +d[9];
}

function isCorpNo(s) {
  const d = String(s).replace(/\D/g, '');
  if (d.length !== 13) return false;
  const sum = [...d.slice(0, 12)].reduce((acc, c, i) => acc + +c * (i % 2 ? 2 : 1), 0);

}

### 주민등록번호 형식 (생년월일 · 성별 자리)

function rrnInfo(s) {
  const d = String(s).replace(/\D/g, '');
  if (!/^\d{13}$/.test(d)) return null;
  const g = +d[6];
  const century = { 1: 1900, 2: 1900, 3: 2000, 4: 2000, 5: 1900, 6: 1900, 7: 2000, 8: 2000, 9: 1800, 0: 1800 }[g];
  const birth = new Date(century + +d.slice(0, 2), +d.slice(2, 4) - 1, +d.slice(4, 6));
  if (birth.getMonth() !== +d.slice(2, 4) - 1) return null;
  return { birth: birth.toLocaleDateString('sv-SE'), male: g % 2 === 1, masked: `${d.slice(0, 6)}-${d[6]}******` };
}

### 여러 칸 규칙을 표로 검사

const rules = {
  code: [[v => !!v, '코드는 필수입니다.'], [v => /^[A-Z]\d{3}$/.test(v), '코드는 영문 대문자 1자 + 숫자 3자입니다.']],
  qty: [[v => Number.isInteger(+v) && +v > 0, '수량은 1 이상 정수입니다.']],
};
function validate(form) {
  const errors = {};
  for (const [field, checks] of Object.entries(rules)) {
    const failed = checks.find(([ok]) => !ok(form[field]));
    if (failed) errors[field] = failed[1];
  }
  return errors;
}

### CSV 글 읽기 (따옴표 안 쉼표 · 줄바꿈)
> 간단한 split(',') 은 따옴표 안 쉼표에서 틀립니다. 한 글자씩 읽습니다.
function parseCsv(text) {
  const rows = [];
  let row = [], cell = '', quoted = false;
  for (let i = 0; i < text.length; i++) {
    const c = text[i];
    if (quoted) {
      if (c === '"' && text[i + 1] === '"') { cell += '"'; i++; }
      else if (c === '"') quoted = false;
      else cell += c;
    } else if (c === '"') quoted = true;
    else if (c === ',') { row.push(cell); cell = ''; }
    else if (c === '\n' || c === '\r') {
      if (c === '\r' && text[i + 1] === '\n') i++;
      row.push(cell); rows.push(row); row = []; cell = '';
    } else cell += c;
  }
  if (cell || row.length) { row.push(cell); rows.push(row); }
  const [head, ...body] = rows;
  return body.map(r => Object.fromEntries(head.map((h, i) => [h, r[i] ?? ''])));
}
console.table(parseCsv('code,name,memo\nA1,"볼트, M8","길이 ""20"""\nA2,너트,"두 줄\n메모"'));

### 가장 먼저 끝난 것 (Promise.race · any)

### 시간 제한 걸기 (타임아웃)
> 응답이 n 초 안에 안 오면 실패로.
function withTimeout(promise, ms) {
  let timer;
  const timeout = new Promise((_, fail) => { timer = setTimeout(() => fail(new Error(`${ms}ms 시간 초과`)), ms); });
  return Promise.race([promise, timeout]).finally(() => clearTimeout(timer));
}
const slow = new Promise(r => setTimeout(() => r('응답'), 500));
try { console.log(await withTimeout(slow, 100)); } catch (e) { console.log(e.message); }
console.log(await withTimeout(Promise.resolve('빠른 응답'), 100));

### 실패하면 다시 시도 (점점 길게 쉬며)
> 네트워크가 잠깐 끊겼을 때. 1번 · 2번 실패 뒤 3번째 성공 예.
async function retry(fn, times = 3, delay = 100) {
  for (let i = 1; ; i++) {
    try {
      return await fn(i);
    } catch (e) {
      if (i >= times) throw e;
      console.log(`${i}번째 실패: ${e.message}, ${delay * i}ms 뒤 다시`);
      await new Promise(r => setTimeout(r, delay * i));
    }
  }
}
console.log(await retry(async n => { if (n < 3) throw new Error('연결 끊김'); return '성공'; }));

### 동시에 n 개까지만 (요청 수 제한)
> 파일 100개를 한꺼번에 올리지 않고 3개씩.
async function mapLimit(items, limit, fn) {
  const results = [];
  let next = 0;
  const worker = async () => {
    while (next < items.length) {
      const i = next++;
      results[i] = await fn(items[i]);
    }
  };
  await Promise.all(Array.from({ length: Math.min(limit, items.length) }, worker));
  return results;
}
let running = 0, peak = 0;
const out = await mapLimit([1, 2, 3, 4, 5, 6, 7], 3, async x => {
  peak = Math.max(peak, ++running);
  await new Promise(r => setTimeout(r, 30));
  running--;
  return x * 10;
});
console.log(out, '동시에 가장 많이', peak);

### 입력 멈춘 뒤 한 번만 (debounce)
> 검색 칸 자동완성: 글자 칠 때마다가 아니라 멈춘 뒤 한 번.
function debounce(fn, ms) {
  let timer;
  return (...args) => { clearTimeout(timer); timer = setTimeout(() => fn(...args), ms); };
}

### 일정 간격에 한 번만 (throttle)
> 스크롤 · 창 크기 바꿈 처리.
function throttle(fn, ms) {
  let last = 0;
  return (...args) => { const now = Date.now(); if (now - last >= ms) { last = now; fn(...args); } };
}
const onScroll = throttle(i => console.log('처리', i), 100);
for (let i = 0; i < 10; i++) {
  onScroll(i);
  await new Promise(r => setTimeout(r, 30));
}

### 이전 요청 무르기 (AbortController)
> 검색어를 바꾸면 앞 요청을 취소해 늦게 온 옛 결과가 화면을 덮지 않게.
function fakeFetch(q, signal) {
  return new Promise((ok, fail) => {
    const t = setTimeout(() => ok(`${q} 결과`), 100);

  });
}

async function search(q) {
  controller?.abort();
  controller = new AbortController();
  try { console.log(await fakeFetch(q, controller.signal)); } catch (e) { if (e.name === 'AbortError') console.log(q, '취소됨'); }
}

await search('볼트');

### 같은 요청 한 번만 (진행 중 공유)

## 유틸
### 한 번만 실행 (once)
> 초기화 함수가 여러 번 불려도 한 번만.
const once = fn => { let done = false, v; return (...a) => (done ? v : ((done = true), (v = fn(...a)))); };
const init = once(() => { console.log('초기화'); return 42; });
console.log(init(), init(), init());

### 결과 기억 (memoize)
> 같은 인자로 다시 부르면 계산하지 않습니다.
const memo = fn => { const cache = new Map(); return n => (cache.has(n) ? cache.get(n) : cache.set(n, fn(n)).get(n)); };
const fib = memo(n => (n < 2 ? n : fib(n - 1) + fib(n - 2)));
console.log(fib(90));

### 함수 이어 붙이기 (pipe)
> 가공 단계를 차례로 적용합니다.
const pipe = (...fns) => x => fns.reduce((v, f) => f(v), x);
const clean = pipe(s => s.trim(), s => s.replace(/\s+/g, ' '), s => s.toUpperCase());
console.log(clean('  a   b  c '));

### 오래 걸리는 반복 나눠 하기 (화면 멈춤 막기)
> 10만 건 처리를 1000건씩 끊고 쉬어 화면이 굳지 않게 합니다.
async function inChunks(items, size, fn) {

    items.slice(i, i + size).forEach(fn);
    await new Promise(r => setTimeout(r, 0));
  }
}

### 조합 함수 (composable) 로 목록 조회 묶기
> 여러 화면에서 쓰는 조회 · 로딩 · 오류 처리를 함수 하나로.
@보기만
import { ref } from 'vue';

export function useList(fetcher) {
  const rows = ref([]);
  const loading = ref(false);
  const error = ref(null);
  async function load(params) {
    loading.value = true;
    error.value = null;
    try {
      rows.value = await fetcher(params);
    } catch (e) {
      error.value = e;
    } finally {
      loading.value = false;
    }
  }
  return { rows, loading, error, load };
}

@보기만
let controller;
async function search() {
  controller?.abort();
  controller = new AbortController();
  try {
    rows.value = (await api.get('/items', { params: cond, signal: controller.signal })).data;
  } catch (e) {
    if (!axios.isCancel(e)) throw e;
  }
}

### 토큰 만료 시 다시 받아 재요청
> 401 이면 새 토큰을 받고 같은 요청을 한 번만 다시 보냅니다.
@보기만
api.interceptors.response.use(null, async err => {
  const original = err.config;
  if (err.response?.status === 401 && !original._retried) {
    original._retried = true;
    const { data } = await axios.post('/api/auth/refresh');
    sessionStorage.setItem('token', data.token);
    original.headers.Authorization = `Bearer ${data.token}`;
    return api(original);
  }
  return Promise.reject(err);
});

### fetch 로 같은 일 (axios 없이)
> fetch 는 4xx · 5xx 에서도 실패하지 않으므로 ok 를 직접 봅니다.
@보기만
async function getJson(url, params) {
  const res = await fetch(`${url}?${new URLSearchParams(params)}`, { headers: { Accept: 'application/json' } });
  if (!res.ok) throw new Error(`${res.status} ${await res.text()}`);
  return res.json();
}

  try {

### 글을 파일로 내려받기 (Blob)
> CSV · JSON 을 서버 없이 파일로.
@보기만
function download(text, name, type = 'text/csv;charset=utf-8') {
  const url = URL.createObjectURL(new Blob(['﻿' + text], { type }));
  const a = Object.assign(document.createElement('a'), { href: url, download: name });
  document.body.append(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}

### 이미지 줄여 올리기 (canvas)
> 휴대폰 사진을 긴 변 1600px 로 줄여 용량을 줄입니다.
@보기만
async function shrink(file, max = 1600) {
  const img = await createImageBitmap(file);
  const scale = Math.min(1, max / Math.max(img.width, img.height));
  const canvas = Object.assign(document.createElement('canvas'), { width: img.width * scale, height: img.height * scale });
  canvas.getContext('2d').drawImage(img, 0, 0, canvas.width, canvas.height);
  return new Promise(r => canvas.toBlob(r, 'image/jpeg', 0.85));
}
